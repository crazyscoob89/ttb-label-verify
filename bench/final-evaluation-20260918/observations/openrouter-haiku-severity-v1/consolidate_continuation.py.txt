  c=sum(x['field_correct'].values());total=7*x['calls'];accuracy=f'{100*c/total:.2f}%' if total else 'unobserved';latency=f"{x['latency_p50_s']:.3f} / {x['latency_p95_s']:.3f}" if x['calls'] else 'unobserved'
  lines.append(f"| {condition} | {x['calls']} | {c}/{total} | {accuracy} | {x['false_matches']}/{x['negative_fields']} | {x['valid_referral_calls']} | {x['processing_failures']} | {latency} |")
 costs=result['costs']
 lines+=['','### Cost and liability (USD)',f"- OpenRouter provider-reported usage.cost: **${observed}**, known for {costs['usage_cost_known_calls']}/{n} requests.  if planned-ids:lines+=['',f"**NOT COMPLETE:** {len(planned-ids)} of {len(planned)} unique requested observations still have no saved result. No additional continuation is authorized or launched."]
