"""Offline one-use continuation fence; no real ledger or network."""
import unittest,tempfile,json,socket
from pathlib import Path
from unittest.mock import patch
import run

class Continuation(unittest.TestCase):
 def setUp(self):
  self.tmp=tempfile.TemporaryDirectory();self.d=Path(self.tmp.name)
  self.net=patch.object(socket.socket,'connect',side_effect=AssertionError('NETWORK FORBIDDEN'));self.net.start()
  self.where=patch.object(run,'D',self.d);self.where.start()
  (self.d/'rows').mkdir();before=self.d/'continuation-1-before';before.mkdir()
  hashes={}
  for i in range(148):
   p=self.d/'rows'/f'{i}.json';p.write_text(json.dumps({'id':str(i),'http_status':502 if i<2 else 200}))
   hashes[str(p.relative_to(self.d))]=run.A.sha(p)
  (before/'preservation.json').write_text(json.dumps({'hashes':hashes}))
  (before/'execution-state.json').write_text(json.dumps({'terminal':True,'reason':'two-transport-failures','saved_unique_rows':148,'unscored_unique_images':64,'attempted_unscored':[]}))
 def tearDown(self):self.where.stop();self.net.stop();self.tmp.cleanup()
 def test_authorized_once_preserves_rows_and_rejects_second_start(self):
  self.assertTrue(hasattr(run,'initial_failures'),'one-use continuation is unsupported')
  hashes={p.name:run.A.sha(p) for p in (self.d/'rows').glob('*.json')}
  self.assertEqual(run.initial_failures(True),0)
  self.assertEqual(hashes,{p.name:run.A.sha(p) for p in (self.d/'rows').glob('*.json')})
  with self.assertRaises(AssertionError):run.initial_failures(True)
  self.assertEqual(run.initial_failures(False),2)
 def test_baseline_drift_denied(self):
  self.assertTrue(hasattr(run,'initial_failures'))
  (self.d/'rows'/'0.json').write_text('{"http_status":200}')
  with self.assertRaises(AssertionError):run.initial_failures(True)
 def test_auth_billing_rate_limit_denied(self):
  self.assertTrue(hasattr(run,'initial_failures'))
  p=self.d/'rows'/'0.json';p.write_text('{"http_status":429}')
  manifest=self.d/'continuation-1-before'/'preservation.json';data=json.loads(manifest.read_text());data['hashes']['rows/0.json']=run.A.sha(p);manifest.write_text(json.dumps(data))
  with self.assertRaises(AssertionError):run.initial_failures(True)
 def test_queue_guard_still_stops_after_two_new_failures(self):
  import ast
  tree=ast.parse(Path(run.__file__).read_text());main=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='main')
  loop=next(n for n in ast.walk(main) if isinstance(n,ast.For) and isinstance(n.iter,ast.Name) and n.iter.id=='queue')
  guard=next(n for n in loop.body if isinstance(n,ast.If) and ast.unparse(n.test)=='failures >= 2')
  for count,expected in [(0,False),(1,False),(2,True)]:self.assertEqual(eval(compile(ast.Expression(guard.test),'<guard>','eval'),{'failures':count}),expected)

if __name__=='__main__':unittest.main(verbosity=2)
