def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def dump(p,v): Path(p).write_text(json.dumps(v,indent=2)+'\n')
def encode(im):
    b=io.BytesIO(); im.save(b,format='PNG',compress_level=6); return b.getvalue()
def glare(src,t,p):
    im=Image.open(src).convert('RGB'); original=np.asarray(im); a=np.asarray(im,dtype=np.float32)/255
    h,w=a.shape[:2]; yy=np.arange(h,dtype=np.float32)[:,None]; xx=np.arange(w,dtype=np.float32)[None,:]
    profile=t['bottle_profile_y_radius']; rad=np.interp(yy[:,0],[v[0] for v in profile],[v[1] for v in profile]).astype(np.float32)[:,None]
    u=(xx-1100)/rad; angle=np.arcsin(np.clip(u,-1,1)); dy=(yy-p['center_y'])/p['sigma_y']; envelope=np.exp(-.5*dy**2)
    center=np.arcsin(p['center_u'])+p['curve']*np.tanh(dy)+.022*dy**2
    width=p['angular_width']*(.60+.40*envelope); delta=angle-center; asym=1+.20*np.tanh(delta/(width+.00001))
    lobe=np.exp(-.5*(delta/(width*asym))**2)*envelope
    halo=np.exp(-.5*(delta/(width*2.2))**2)*np.exp(-.5*(dy/1.24)**2)
    edge_center=-.88 if p['center_u']>0 else .88
    edge=np.exp(-.5*((u-edge_center)/.038)**2)*np.exp(-.5*((yy-(p['center_y']+100))/650)**2)
    silhouette=np.clip((1-np.abs(u))/.035,0,1)*np.clip((yy-310)/170,0,1)*np.clip((3020-yy)/100,0,1)
    alpha=(p['strength']*((1-p['bloom_fraction'])*lobe+p['bloom_fraction']*halo)+p['edge_strength']*edge)*silhouette
    uncapped_max=float(alpha.max()); clipped_pixels=int((alpha>.65).sum())
    alpha=np.minimum(alpha,np.float32(.65))
    linear=np.where(a<=.04045,a/12.92,((a+.055)/1.055)**2.4)
    lit=linear*(1-alpha[...,None])+np.array([1.,.985,.955],np.float32)*alpha[...,None]
    out=np.where(lit<=.0031308,12.92*lit,1.055*np.maximum(lit,0)**(1/2.4)-.055)
    out=np.rint(np.clip(out,0,1)*255).astype(np.uint8); out[alpha<1e-7]=original[alpha<1e-7]
    la=alpha[1100:2945,500:1701]; lo=out[1100:2945,500:1701]; old=original[1100:2945,500:1701]
    sat=np.all(lo>=254,axis=2); dark=np.max(old,axis=2)<100; newly=sat&~np.all(old>=254,axis=2)
    delta=np.max(np.abs(out.astype(np.int16)-original.astype(np.int16)),axis=2)
    stats=dict(alpha_max=float(alpha.max()),source_weight_min=float(1-alpha.max()),max_adjacent_alpha_step=float(max(np.abs(np.diff(alpha,axis=0)).max(),np.abs(np.diff(alpha,axis=1)).max())),cap_alpha_max=float(alpha[:310].max()),base_alpha_max=float(alpha[3020:].max()),outside_bottle_changed_pixels=int(((silhouette==0)&(delta>0)).sum()),label_max_row_fraction_alpha_above_020=float((la>.20).mean(axis=1).max()),label_max_row_fraction_alpha_above_050=float((la>.50).mean(axis=1).max()),solid_saturated_full_label_columns=int(sat.all(axis=0).sum()),solid_saturated_full_label_rows=int(sat.all(axis=1).sum()),new_saturated_dark_ink_pixels=int((sat&dark).sum()),max_row_new_saturated_fraction=float(newly.mean(axis=1).max()),label_mean_absolute_rgb_change=float(delta[1100:2945,500:1701].mean()))
    stats.update(requested_intensity_gain_from_v1=GAIN,uncapped_alpha_max=uncapped_max,opacity_cap=.65,opacity_clipped_pixels=clipped_pixels,effective_peak_gain_from_v1=float(alpha.max())/(uncapped_max/GAIN))
    assert stats['alpha_max']<=.65000001 and stats['source_weight_min']>=.34999999
    assert stats['max_adjacent_alpha_step']<.025
    assert stats['cap_alpha_max']==stats['base_alpha_max']==stats['outside_bottle_changed_pixels']==0
    assert stats['solid_saturated_full_label_columns']==stats['solid_saturated_full_label_rows']==stats['new_saturated_dark_ink_pixels']==0
    assert stats['max_row_new_saturated_fraction']<.01
    assert stats['label_max_row_fraction_alpha_above_020']<.55 and stats['label_max_row_fraction_alpha_above_050']<.30
    assert stats['label_mean_absolute_rgb_change']>1
    return Image.fromarray(out),stats

def render(effect,row):
    if effect=='glare': return glare(row['source_path'],row['source_geometry'],row['parameters'])
    im=Image.open(row['source_path']).convert('RGB').filter(ImageFilter.GaussianBlur(row['parameters']['sigma_native_px']))
    return im,dict(filter='GaussianBlur',sigma_native_px=SIGMA,falloff='Spatially uniform symmetric normalized Gaussian approximation; no bands or masks added; no finite localized glare field applicable')

def boards(grows,brows):
    g={r['fixture_id']:r for r in grows}; b={r['fixture_id']:r for r in brows}
    font=lambda n:ImageFont.truetype('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf',n)
    def board(path,title,rows,crop,panel):
        cols=len(rows[0][1]); pw,ph=panel; W=cols*(pw+16)+16; H=100+len(rows)*(ph+58)
        im=Image.new('RGB',(W,H),'#f5f2ec');d=ImageDraw.Draw(im)
        d.text((16,10),title,font=font(29),fill='black');d.text((16,50),'PREVIEW ONLY - not approved / not model-tested',font=font(22),fill='#555555')
        for k,(label,cells) in enumerate(rows):
            y=100+k*(ph+58)
            for j,(heading,p) in enumerate(cells):
                x=16+j*(pw+16);d.text((x,y),heading,font=font(26),fill='black')
                im.paste(Image.open(p).crop(crop).resize(panel,Image.Resampling.LANCZOS),(x,y+42))
        im.save(path,quality=95,subsampling=0)
    ident='C-SPIRITS-01'; gr=g[ident];br=b[ident]
    # Label-wide overview plus native-scale detail rows makes effect comparison phone-friendly when opened.
    board(BLUR/'C-SPIRITS-01-three-way.jpg','C-SPIRITS-01 | Original / glare / blur',[(ident,[('Original',gr['source_path']),('Glare ×2.5',GLARE/gr['output']),('Blur sigma5.5',BLUR/br['output'])])],(470,1060,1730,2970),(480,728))
    board(GLARE/'selected3-pairs.jpg','Glare V3 | requested x2.5; opacity capped 0.65',[(i,[('Original '+i,g[i]['source_path']),('Glare ×2.5',GLARE/g[i]['output'])]) for i in ['C-SPIRITS-01','C-WINE-01','C-MALT-01']],(470,1060,1730,2970),(440,667))
